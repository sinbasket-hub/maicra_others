Outlookメール自動送信設定（GAS連携）

打刻・退勤のたびにGoogle Apps Script経由でOutlookへ控えメールを自動送信します。

GAS側のセットアップ手順（初回のみ・約2分）

  1. Googleドライブで「新規」＞「その他」＞「Google Apps Script」を作成します。
  2. 下のコードをすべてコピーして貼り付け、保存します。
  3. 右上の「デプロイ」＞「新しいデプロイ」から種類を「ウェブアプリ」に設定します。
  4. アクセスできるユーザーを「全員」にしてデプロイし、発行されたURLを上の欄に貼り付けてください。
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var toEmail = data.email;
    var type = data.type;
    var time = data.time;
    var note = data.note || "なし";
    var userName = data.userName || "未設定";

    var subject = "【勤怠控え】" + type + "打刻のお知らせ（" + time + "）";
    var body = "お疲れさまです。\n"
             + "以下の内容で勤怠打刻が完了しました。\n\n"
             + "■ 氏名：" + userName + "\n"
             + "■ 種別：" + type + "\n"
             + "■ 日時：" + time + "\n"
             + "■ 備考：" + note + "\n\n"
             + "----------------------------------\n"
             + "スマート勤怠管理システム 自動送信";

    MailApp.sendEmail(toEmail, subject, body);

    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
